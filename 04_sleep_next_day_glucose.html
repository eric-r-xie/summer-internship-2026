# sleep features
sleep_path <- "data/mini/sleep/garmin_vivosmart5/1029/1029_sleep.json"
sleep_json <- fromJSON(sleep_path)

stages <- sleep_json$body$sleep |>
  unnest_wider(effective_time_frame) |>
  unnest_wider(time_interval)

sleep_tbl <- stages |>
  mutate(participant_id = sleep_json$header$user_id) |>
  mutate(
    start = ymd_hms(start_date_time, tz = "UTC"),
    end   = ymd_hms(end_date_time, tz = "UTC"),
    duration_min = as.numeric(difftime(end, start, units = "mins")),
    sleep_night = as_date(if_else(hour(start) < 18, start - days(1), start))
  ) |>
  group_by(participant_id, sleep_night) |>
  summarise(
    bedtime               = min(start),
    wake_time              = max(end),
    total_sleep_time_min  = sum(duration_min[sleep_stage_state != "awake"]),
    time_in_bed_min        = as.numeric(difftime(max(end), min(start), units = "mins")),
    awake_after_onset_min  = sum(duration_min[sleep_stage_state == "awake"]),
    pct_light = sum(duration_min[sleep_stage_state == "light"]) / sum(duration_min) * 100,
    pct_deep  = sum(duration_min[sleep_stage_state == "deep"])  / sum(duration_min) * 100,
    pct_rem   = sum(duration_min[sleep_stage_state == "rem"])   / sum(duration_min) * 100,
    pct_awake = sum(duration_min[sleep_stage_state == "awake"]) / sum(duration_min) * 100,
    .groups = "drop"
  ) |>
  mutate(sleep_efficiency_pct = round(total_sleep_time_min / time_in_bed_min * 100, 2)) |>
  arrange(sleep_night)

# Daily activity features (steps, calories, heart rate)
pa_path  <- "data/mini/physical_activity/garmin_vivosmart5/1029/1029_activity.json"
pac_path <- "data/mini/physical_activity_calorie/garmin_vivosmart5/1029/1029_calorie.json"
hr_path  <- "data/mini/heart_rate/garmin_vivosmart5/1029/1029_heartrate.json"

pa_json  <- fromJSON(pa_path)
pac_json <- fromJSON(pac_path)
hr_json  <- fromJSON(hr_path)

# steps
steps_raw <- pa_json$body$activity |>
  unnest_wider(base_movement_quantity, names_sep = "_") |>
  unnest_wider(effective_time_frame) |>
  unnest_wider(time_interval)

steps_daily <- steps_raw |>
  mutate(
    date  = as_date(ymd_hms(start_date_time, tz = "UTC")),
    steps = as.numeric(base_movement_quantity_value)
  ) |>
  group_by(date) |>
  summarise(total_steps = sum(steps, na.rm = TRUE), .groups = "drop")

# active calories
cal_raw <- pac_json$body$activity |>
  unnest_wider(calories_value, names_sep = "_") |>
  unnest_wider(effective_time_frame)

cal_daily <- cal_raw |>
  mutate(date = as_date(ymd_hms(date_time, tz = "UTC"))) |>
  group_by(date) |>
  summarise(active_calories = sum(calories_value_value, na.rm = TRUE), .groups = "drop")

# heart rate
hr_raw <- hr_json$body$heart_rate |>
  unnest_wider(heart_rate, names_sep = "_") |>
  unnest_wider(effective_time_frame)

hr_daily <- hr_raw |>
  mutate(date = as_date(ymd_hms(date_time, tz = "UTC"))) |>
  filter(heart_rate_value != 0) |>
  group_by(date) |>
  summarise(average_heartrate_bpm = mean(heart_rate_value, na.rm = TRUE),
            .groups = "drop")

daily_activity_tbl <- steps_daily |>
  full_join(cal_daily, by = "date") |>
  full_join(hr_daily, by = "date") |>
  mutate(participant_id = pa_json$header$user_id, .before = 1) |>
  arrange(date)

# CGM daily features
glucose_path <- "data/mini/dexcom_g6/1029/1029_DEX.json"
glucose_json <- fromJSON(glucose_path)

glucose_raw <- glucose_json$body$cgm |>
  unnest_wider(blood_glucose, names_sep = "_") |>
  unnest_wider(effective_time_frame) |>
  unnest_wider(time_interval)

glucose_tbl <- glucose_raw |>
  mutate(reading_time = ymd_hms(start_date_time, tz = "UTC")) |>
  select(reading_time, glucose_mg_dl = blood_glucose_value)

glucose_daily <- glucose_tbl |>
  mutate(date = as_date(reading_time)) |>
  group_by(date) |>
  summarise(
    mean_glucose_mg_dl = mean(glucose_mg_dl, na.rm = TRUE),
    sd_glucose_mg_dl = sd(glucose_mg_dl, na.rm = TRUE),
    n_glucose_readings = n(),
    .groups = "drop"
  )

# link all
person_night_tbl <- sleep_tbl |>
  mutate(
    activity_date = sleep_night + days(1),
    cgm_date      = sleep_night + days(1)
  ) |>
  left_join(daily_activity_tbl, by = c("participant_id", "activity_date" = "date")) |>
  left_join(glucose_daily,      by = c("cgm_date" = "date")) |>
  arrange(sleep_night)
